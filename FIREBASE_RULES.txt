
// Firestore Rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    function canAccessGarden(gardenId) {
      let garden = get(/databases/$(database)/documents/gardens/$(gardenId)).data;
      return garden.ownerId == request.auth.uid || request.auth.uid in garden.collaboratorIds;
    }

    function isGardenOwner(gardenId) {
      let garden = get(/databases/$(database)/documents/gardens/$(gardenId)).data;
      return garden.ownerId == request.auth.uid;
    }

    match /gardens/{gardenId} {
      allow create: if request.auth != null;
      allow read: if request.auth != null && (resource.data.ownerId == request.auth.uid || request.auth.uid in resource.data.collaboratorIds);
      allow update: if request.auth != null && (resource.data.ownerId == request.auth.uid || request.auth.uid in resource.data.collaboratorIds);
      allow delete: if request.auth != null && resource.data.ownerId == request.auth.uid;
    }

    match /photos/{photoId} {
      allow read: if request.auth != null && canAccessGarden(resource.data.gardenId);
      allow create: if request.auth != null && canAccessGarden(request.resource.data.gardenId);
      allow delete: if request.auth != null && canAccessGarden(resource.data.gardenId);
    }

    match /users/{userId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
  }
}

// Storage Rules
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /gardens/{userId}/{allPaths=**} {
       allow read: if true;
       allow write: if request.auth != null && request.auth.uid == userId
                    ||
                    firestore.get(/databases/(default)/documents/users/$(userId)).data.editKey == request.resource.metadata.editKey;
    }
  }
}
